17 nda::array<dcomplex, 3> get_extended_coefficients(
18 nda::array_const_view<dcomplex, 3> hyb_coeffs,
19 nda::array_const_view<dcomplex, 3> dynint_coeffs) {
20
21 assert( hyb_coeffs.extent(0) == dynint_coeffs.extent(0) );
22 assert( hyb_coeffs.extent(1) == hyb_coeffs.extent(2) );
23 assert( dynint_coeffs.extent(1) == dynint_coeffs.extent(2) );
24
25 int r = hyb_coeffs.extent(0);
26 int n_hyb = hyb_coeffs.extent(1);
27 int n_dynint = dynint_coeffs.extent(1);
28 int n_ext = n_hyb + n_dynint;
29
30 auto ext_coeffs = nda::array<dcomplex, 3>::zeros(r, n_ext, n_ext);
31
32 ext_coeffs(_, range(0, n_hyb), range(0, n_hyb)) = hyb_coeffs;
33 ext_coeffs(_, range(n_hyb, n_ext), range(n_hyb, n_ext)) = dynint_coeffs;
34
35 return ext_coeffs;
36 }

116 struct DynintOpBlocks {
117 nda::vector<int> connection; // connection row of O, -1 where the block is absent
118 nda::vector<int> connection_dag; // connection row of O^dag, the inverse map
119 std::vector<nda::array<dcomplex, 2>> blocks; // Fock-basis blocks of O
120 std::vector<nda::array<dcomplex, 2>> blocks_dag; // Fock-basis blocks of O^dag
121 };

123 template <bool IsComplex>
124 DynintOpBlocks resolve_dynint_op(const triqs_atom_diag_t<IsComplex> &ad, triqs::operators::many_body_operator_real const &op, int i) {
125
126 int nb = ad.n_subspaces();
127
128 auto context = [&](std::string const &what) {
129 std::ostringstream msg;
130 msg << "get_operators_and_interactions: dynamical interaction operator " << i << " (" << op << ") " << what;
131 return msg.str();
132 };
133
134 // get_op_mat throws when two monomials of op take one subspace to different targets.
135 auto op_mat_of = [&](triqs::operators::many_body_operator_real const &o, std::string const &what) {
136 try {
137 return ad.get_op_mat(o);
138 } catch (std::exception const &e) {
139 throw std::invalid_argument(context(what + ": atom_diag::get_op_mat failed with \"" + std::string(e.what())
140 + "\". Every monomial must take each atom_diag subspace to the same target subspace."));
141 }
142 };
143
144 auto om = op_mat_of(op, "is not block resolvable");
145 auto conn = nda::vector<int>(nb);
146 for (int b = 0; b < nb; ++b) conn(b) = static_cast<int>(om.connection(b));
147
148 // an operator that is zero on every subspace would give a symmetry set of size zero
149 bool any_present = false;
150 for (int b = 0; b < nb; ++b) any_present = any_present || (conn(b) != -1);
151 if (!any_present) throw std::invalid_argument(context("is zero on every atom_diag subspace"));
152
153 // The connection row must be injective for the adjoint to have a well defined connection row. Checked before
154 // the adjoint is fetched, since get_op_mat(dagger(op)) would otherwise throw from inside TRIQS with a less specific message.
155 std::set<int> seen;
156 for (int b = 0; b < nb; ++b) {
157 if (conn(b) == -1) continue;
158 if (!seen.insert(conn(b)).second) {
159 int b_first = -1;
160 for (int a = 0; a < b; ++a) {
161 if (conn(a) == conn(b)) {
162 b_first = a;
163 break;
164 }
165 }
166 throw std::invalid_argument(context("has a non-injective connection map: subspaces " + std::to_string(b_first) + " and "
167 + std::to_string(b) + " are both taken to subspace " + std::to_string(conn(b))
168 + ". The block-sparse storage keeps one block per block column and inverts the map to "
169 "store the adjoint, so this operator cannot be represented."));
170 }
171 }
172
173 auto omd = op_mat_of(triqs::operators::many_body_operator_real{dagger(op)}, "has an adjoint that is not block resolvable");
174 auto conn_dag = nda::vector<int>(nb);
175 for (int b = 0; b < nb; ++b) conn_dag(b) = static_cast<int>(omd.connection(b));
176
177 // the connection row of the adjoint is the inverse of the injective row
178 for (int b = 0; b < nb; ++b) {
179 if (conn(b) != -1 && conn_dag(conn(b)) != b)
180 throw std::invalid_argument(context("has an adjoint whose connection map is not the inverse of its own"));
181 if (conn_dag(b) != -1 && conn(conn_dag(b)) != b)
182 throw std::invalid_argument(context("has an adjoint whose connection map is not the inverse of its own"));
183 }
184
185 // Rotate the eigenbasis blocks to the Fock basis, as setup_ops_from_triqs_2nd_quant_ops does.
186 auto rotate = [&](auto const &m, nda::vector_const_view<int> cn) {
187 std::vector<nda::array<dcomplex, 2>> out(nb);
188 for (int b = 0; b < nb; ++b) {
189 if (cn(b) == -1) {
190 out[b] = nda::zeros<dcomplex>(1, 1);
191 continue;
192 }
193 auto UR = ad.get_unitary_matrix(b);
194 auto UL = ad.get_unitary_matrix(cn(b));
195 out[b] = nda::array<dcomplex, 2>{UL * m.block_mat[b] * nda::conj(nda::transpose(UR))};
196 }
197 return out;
198 };
199
200 return {conn, conn_dag, rotate(om, conn), rotate(omd, conn_dag)};
201 }

205 template <bool IsComplex>
206 std::tuple<BlockOpSymQuartet, nda::vector<long>> get_operators_and_interactions_impl(
207 const triqs_atom_diag_t<IsComplex> &ad,
208 nda::array_const_view<dcomplex, 3> hyb_coeffs,
209 nda::array_const_view<dcomplex, 3> dynint_coeffs,
210 std::vector<triqs::operators::many_body_operator_real> const &dynint_ops) {
211
212 int nb = ad.n_subspaces();
213 int p = hyb_coeffs.extent(0);
214 int n_hyb = hyb_coeffs.extent(1);
215 int n_int = static_cast<int>(dynint_ops.size());
216
217 if (hyb_coeffs.extent(2) != n_hyb)
218 throw std::invalid_argument("get_operators_and_interactions: hyb_coeffs must have shape (p, n_hyb, n_hyb), got ("
219 + std::to_string(p) + ", " + std::to_string(n_hyb) + ", " + std::to_string(hyb_coeffs.extent(2)) + ")");
220 if (n_hyb != static_cast<int>(ad.get_fops().size()))
221 throw std::invalid_argument("get_operators_and_interactions: hyb_coeffs covers " + std::to_string(n_hyb)
222 + " flavours but the atom_diag has " + std::to_string(ad.get_fops().size())
223 + " fundamental operators");
224 if (dynint_coeffs.extent(1) != n_int || dynint_coeffs.extent(2) != n_int)
225 throw std::invalid_argument("get_operators_and_interactions: dynint_coeffs must have shape (p, n_int, n_int) with n_int = "
226 + std::to_string(n_int) + ", got (" + std::to_string(dynint_coeffs.extent(0)) + ", "
227 + std::to_string(dynint_coeffs.extent(1)) + ", " + std::to_string(dynint_coeffs.extent(2)) + ")");
228 if (dynint_coeffs.extent(0) != p)
229 throw std::invalid_argument("get_operators_and_interactions: hyb_coeffs and dynint_coeffs must share the pole set, got "
230 + std::to_string(p) + " and " + std::to_string(dynint_coeffs.extent(0)) + " poles");
231
232 // fermionic symmetry sets, as in get_operators()
233 auto sets = atom_diag::get_operator_sym_sets(ad, n_hyb);
234 int n_sym_hyb = static_cast<int>(sets.Fs.size());
235
236 // Resolve every interaction operator before grouping, so that a bad one is reported by index.
237 std::vector<DynintOpBlocks> resolved;
238 resolved.reserve(n_int);
239 for (int i = 0; i < n_int; ++i) resolved.push_back(resolve_dynint_op(ad, dynint_ops[i], i));
240
241 // group the interaction operators by identical connection row, i.e. by sparsity pattern, in order of first appearance
242 std::vector<int> group_of_op(n_int, -1);
243 std::vector<std::vector<int>> ops_of_group;
244 for (int i = 0; i < n_int; ++i) {
245 for (int g = 0; g < static_cast<int>(ops_of_group.size()) && group_of_op[i] < 0; ++g) {
246 bool same = true;
247 for (int b = 0; b < nb && same; ++b) same = (resolved[i].connection(b) == resolved[ops_of_group[g][0]].connection(b));
248 if (same) {
249 group_of_op[i] = g;
250 ops_of_group[g].push_back(i);
251 }
252 }
253 if (group_of_op[i] < 0) {
254 group_of_op[i] = static_cast<int>(ops_of_group.size());
255 ops_of_group.push_back({i});
256 }
257 }
258
259 // one BlockOpSymSet per group, with the operators in increasing orbital order as assumed by BlockOpSymQuartet
260 for (auto const &group : ops_of_group) {
261 int qg = static_cast<int>(group.size());
262 auto const &rep = resolved[group[0]];
263
264 std::vector<nda::array<dcomplex, 3>> blocks(nb), blocks_dag(nb);
265 for (int b = 0; b < nb; ++b) {
266 blocks[b] = rep.connection(b) == -1 ? nda::zeros<dcomplex>(qg, 1, 1)
267 : nda::zeros<dcomplex>(qg, rep.blocks[b].shape(0), rep.blocks[b].shape(1));
268 blocks_dag[b] = rep.connection_dag(b) == -1 ? nda::zeros<dcomplex>(qg, 1, 1)
269 : nda::zeros<dcomplex>(qg, rep.blocks_dag[b].shape(0), rep.blocks_dag[b].shape(1));
270 }
271 for (int k = 0; k < qg; ++k) {
272 auto const &rk = resolved[group[k]];
273 for (int b = 0; b < nb; ++b) {
274 if (rep.connection(b) != -1) blocks[b](k, range::all, range::all) = rk.blocks[b];
275 if (rep.connection_dag(b) != -1) blocks_dag[b](k, range::all, range::all) = rk.blocks_dag[b];
276 }
277 }
278 sets.Fs.emplace_back(rep.connection, blocks);
279 sets.F_dags.emplace_back(rep.connection_dag, blocks_dag);
280 }
281
282 // concatenate the labels, the interaction sets follow the fermionic ones
283 int n_ext = n_hyb + n_int;
284 auto labels = nda::vector<long>(n_ext);
285 for (int i = 0; i < n_hyb; ++i) labels(i) = sets.sym_set_labels(i);
286 for (int i = 0; i < n_int; ++i) labels(n_hyb + i) = n_sym_hyb + group_of_op[i];
287
288 // the cross-set guard of the quartet constructor validates the extended coefficients uniformly
289 auto ext_coeffs = get_extended_coefficients(hyb_coeffs, dynint_coeffs);
290
291 return {BlockOpSymQuartet(sets.Fs, sets.F_dags, ext_coeffs, labels), labels};
292 }
