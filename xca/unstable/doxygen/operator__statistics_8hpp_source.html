24 inline bool is_fermionic_parity(triqs::operators::many_body_operator_real const &op, std::string const &ctx) {
25 std::optional<bool> odd;
26 for (auto const &term : op) {
27 bool this_odd = (term.monomial.size() % 2 == 1);
28 if (!odd)
29 odd = this_odd;
30 else if (*odd != this_odd)
31 // an operator of mixed parity has no statistics, while c + c^dag c^dag c is all-odd and well defined
32 throw std::runtime_error(ctx + ": operator has mixed fermion parity (both even- and odd-length monomials), "
33 "so its statistics is undefined");
34 }
35 // no terms at all means the zero operator, the identity has one empty monomial of even length
36 if (!odd) throw std::runtime_error(ctx + ": operator is identically zero, so its statistics is undefined");
37 return *odd;
38 }
