133 // Check the shape before either branch: the assign branch would silently resize the block, and the accumulate branch reads past the

443 void check_sym_set_block_diagonal(nda::array_const_view<dcomplex, 3> hyb_coeffs, nda::vector_const_view<long> sym_set_labels,
444 std::string const &who) {
445
446 // Reject hybridization coefficients that couple different symmetry sets, since a barred operator is stored with the block-sparsity pattern
447 // of its own symmetry set and can only hold contributions from operators of that set. The threshold is relative and one-sided, and the
448 // accepted remainder below it is made inert by the p_mu == p_kap restriction in find_path_self_energy(), not by this check.
449 long n = sym_set_labels.size();
450 int p = hyb_coeffs.extent(0);
451 double max_abs = 0.0;
452 for (int l = 0; l < p; l++) {
453 for (long i = 0; i < n; i++) {
454 for (long j = 0; j < n; j++) { max_abs = std::max(max_abs, std::abs(hyb_coeffs(l, i, j))); }
455 }
456 }
457
458 // Inf and NaN entries would pass the threshold comparison below
459 if (!std::isfinite(max_abs)) { throw std::invalid_argument(who + ": hyb_coeffs contains a non-finite entry"); }
460
461 double thresh = sym_set_coupling_tol * max_abs; // zero for all-zero coefficients
462 double worst = 0.0;
463 long worst_l = -1, worst_i = -1, worst_j = -1;
464 long n_bad = 0, n_cross = 0;
465 for (int l = 0; l < p; l++) {
466 for (long i = 0; i < n; i++) {
467 for (long j = 0; j < n; j++) {
468 if (sym_set_labels(i) == sym_set_labels(j)) continue;
469 n_cross++;
470 double a = std::abs(hyb_coeffs(l, i, j));
471 if (a > thresh) {
472 n_bad++;
473 if (a > worst) {
474 worst = a;
475 worst_l = l;
476 worst_i = i;
477 worst_j = j;
478 }
479 }
480 }
481 }
482 }
483 if (n_bad > 0) {
484 std::ostringstream msg;
485 msg << std::scientific;
486 msg << who << ": hyb_coeffs couples different symmetry sets, which the symmetry-set storage cannot represent. " << n_bad << " of " << n_cross
487 << " cross-set entries exceed the tolerance " << sym_set_coupling_tol << " * max|hyb_coeffs| = " << thresh << "; the largest is "
488 << "|hyb_coeffs(" << worst_l << ", " << worst_i << ", " << worst_j << ")| = " << worst << " (relative " << worst / max_abs
489 << "), coupling symmetry set " << sym_set_labels(worst_i) << " to symmetry set " << sym_set_labels(worst_j) << ".";
490 throw std::invalid_argument(msg.str());
491 }
492 }

496 BlockOpSymQuartet::BlockOpSymQuartet(std::vector<BlockOpSymSet> Fs, std::vector<BlockOpSymSet> F_dags,
497 nda::array_const_view<dcomplex, 3> hyb_coeffs, nda::vector_const_view<long> sym_set_labels)
498 : Fs(Fs), F_dags(F_dags), sym_set_labels(sym_set_labels), p(hyb_coeffs.extent(0)) {
499
500 // Fs and F_dags are vectors of BOSS
501 // Each entry corresponds to a set of operators with the same block-sparse structure
502 // Fs and F_dags have the same number of entries
503 // if k = # of entries of Fs, and each entry f_i has q_i operators, then n = sum(q_i) = number of orbital indices
504 int k = Fs.size();
505 if (k != F_dags.size()) { throw std::invalid_argument("Fs and F_dags must have the same number of entries"); }
506
507 long n = sym_set_labels.size(); // number of orbital indices
508 if (hyb_coeffs.extent(1) != n || hyb_coeffs.extent(2) != n) {
509 throw std::invalid_argument("BlockOpSymQuartet: hyb_coeffs must have shape (p, n, n) with n = sym_set_labels.size() = " + std::to_string(n)
510 + ", got (" + std::to_string(hyb_coeffs.extent(0)) + ", " + std::to_string(hyb_coeffs.extent(1)) + ", "
511 + std::to_string(hyb_coeffs.extent(2)) + ")");
512 }
513
514 // initialize F_dag_bars and F_bars_refl
515 p = hyb_coeffs.extent(0);
516 nda::vector<int> block_indices_dag = F_dags[0].get_block_indices();
517 nda::vector<int> block_indices_f = Fs[0].get_block_indices();
518 for (int i = 0; i < k; i++) {
519 block_indices_dag = F_dags[i].get_block_indices();
520 block_indices_f = Fs[i].get_block_indices();
521 this->F_dag_bars.emplace_back(F_dags[i].get_size_sym_set(), p, block_indices_dag, F_dags[i].get_block_sizes());
522 this->F_bars_refl.emplace_back(Fs[i].get_size_sym_set(), p, block_indices_f, Fs[i].get_block_sizes());
523 }
524
525 // calculate symmetry set indices
526 long q = nda::max_element(sym_set_labels) + 1; // number of symmetry sets
527 sym_set_inds = nda::zeros<long>(n); // indices of orbital indices in symmetry sets
528 sym_set_sizes = nda::zeros<long>(q); // sizes of symmetry sets
529 sym_set_to_orb = nda::ones<long>(q, n); // map label and index to orbital index
530 sym_set_to_orb *= -1; // initialize with -1
531 for (int i = 0; i < n; i++) {
532 sym_set_inds(i) = sym_set_sizes(sym_set_labels(i));
533 sym_set_sizes(sym_set_labels(i))++;
534 sym_set_to_orb(sym_set_labels(i), sym_set_inds(i)) = i; // map symmetry set index to backbone orbital index
535 }
536
537 // the number of symmetry sets must match the number of operator sets
538 if (q != k) {
539 throw std::invalid_argument("BlockOpSymQuartet: sym_set_labels reaches " + std::to_string(q) + " symmetry sets but " + std::to_string(k)
540 + " BlockOpSymSet entries were given; every label must name one of Fs[0..Fs.size())");
541 }
542
543 // the symmetry set sizes from the labels must agree with the operator sets
544 for (int i = 0; i < k; i++) {
545 if (sym_set_sizes(i) != Fs[i].get_size_sym_set() || sym_set_sizes(i) != F_dags[i].get_size_sym_set()) {
546 throw std::invalid_argument("BlockOpSymQuartet: symmetry set " + std::to_string(i) + " has " + std::to_string(sym_set_sizes(i))
547 + " orbital indices according to sym_set_labels, but Fs[" + std::to_string(i) + "] holds "
548 + std::to_string(Fs[i].get_size_sym_set()) + " and F_dags[" + std::to_string(i) + "] holds "
549 + std::to_string(F_dags[i].get_size_sym_set()) + " operators");
550 }
551 }
552
553 check_sym_set_block_diagonal(hyb_coeffs, sym_set_labels, "BlockOpSymQuartet");
554
555 // compute F_dag_bars and F_bars_refl, contracting within each symmetry set only
556 for (int l = 0; l < p; l++) {
557 for (int p_set = 0; p_set < q; p_set++) {
558 for (int lam = 0; lam < sym_set_sizes(p_set); lam++) {
559 for (int nu = 0; nu < sym_set_sizes(p_set); nu++) {
560 long lam_orb = sym_set_to_orb(p_set, lam);
561 long nu_orb = sym_set_to_orb(p_set, nu);
562 for (int b = 0; b < F_dags[p_set].get_num_block_cols(); b++) {
563 if (F_dags[p_set].get_block_index(b) != -1) {
564 F_dag_bars[p_set].add_block(b, lam, l, nda::make_regular(hyb_coeffs(l, nu_orb, lam_orb) * F_dags[p_set].get_block(b)(nu, _, _)));
565 }
566 }
567 for (int b = 0; b < Fs[p_set].get_num_block_cols(); b++) {
568 if (Fs[p_set].get_block_index(b) != -1) {
569 F_bars_refl[p_set].add_block(
570 b, nu, l, nda::make_regular(-hyb_coeffs(l, nu_orb, lam_orb) * Fs[p_set].get_block(b)(lam, _, _))); // Add -1 sign for reflected F
571 }
572 }
573 }
574 }
575 }
576 }
577 }

643 // Evaluate F^dagger in block-sparse storage
644 // @param[in] F F operator
645 // @return F^dagger operator

648

710 BlockDiagOpFun BOFtoBDOF(BlockOpFun const &A) {
711 // Convert a BlockOpFun with diagonal structure to a BlockDiagOpFun
712 // @param[in] A BlockOpFun
713 // @return BlockDiagOpFun
714
715 int num_block_cols = A.get_num_block_cols();
716 auto diag_blocks = A.get_blocks();
717 auto zero_block_indices = nda::zeros<int>(num_block_cols);
718 for (int i = 0; i < num_block_cols; i++) {
719 int block_index = A.get_block_index(i);
720 if (block_index == -1) {
721 diag_blocks[i] = nda::zeros<dcomplex>(1, 1, 1);
722 zero_block_indices(i) = -1;
723 } else if (block_index != i) {
724 throw std::invalid_argument("BOF is not diagonal");
725 }
726 }
727
728 return BlockDiagOpFun(diag_blocks, zero_block_indices);

731 BlockDiagOpFun nonint_gf_BDOF(std::vector<nda::array<dcomplex, 2>> H_blocks, nda::vector<int> H_block_inds, double beta,
732 nda::vector_const_view<double> dlr_it_abs) {
733
734 int num_block_cols = H_block_inds.size();
735 nda::vector<int> H_block_sizes(num_block_cols);
736 for (int i = 0; i < num_block_cols; i++) { H_block_sizes(i) = H_blocks[i].extent(0); }
737
738 int r = dlr_it_abs.size();
739
740 dcomplex tr_exp_minusbetaH = 0;
741 std::vector<nda::array<dcomplex, 1>> H_evals(num_block_cols);
742 std::vector<nda::array<dcomplex, 2>> H_evecs(num_block_cols);
743 for (int i = 0; i < num_block_cols; i++) {
744 if (H_block_inds(i) != -1) {
745 if (H_block_sizes(i) == 1) {
746 H_evals[i] = nda::array<dcomplex, 1>{H_blocks[i](0, 0)};
747 H_evecs[i] = nda::array<dcomplex, 2>{{1}};
748 } else {
749 auto H_block_eig = nda::linalg::eigh(H_blocks[i]);
750 H_evals[i] = std::get<0>(H_block_eig);
751 H_evecs[i] = std::get<1>(H_block_eig);
752 }
753 tr_exp_minusbetaH += nda::sum(exp(-beta * H_evals[i]));
754 } else {
755 H_evals[i] = nda::zeros<dcomplex>(H_block_sizes(i));
756 H_evecs[i] = nda::eye<dcomplex>(H_block_sizes(i));
757 tr_exp_minusbetaH += 1.0 * H_block_sizes(i); // 0 entry in the diagonal
758 }
759 }
760
761 auto eta_0 = nda::log(tr_exp_minusbetaH) / beta;
762 auto Gt = BlockDiagOpFun(r, H_block_sizes);
763 for (int i = 0; i < num_block_cols; i++) {
764 auto Gt_block = nda::array<dcomplex, 3>(r, H_block_sizes(i), H_block_sizes(i));
765 auto Gt_temp = nda::make_regular(0 * H_blocks[i]);
766 for (int t = 0; t < r; t++) {
767 for (int j = 0; j < H_block_sizes(i); j++) { Gt_temp(j, j) = -exp(-beta * dlr_it_abs(t) * (H_evals[i](j) + eta_0)); }
768 Gt_block(t, _, _) = matmul(H_evecs[i], matmul(Gt_temp, nda::transpose(H_evecs[i])));
769 }
770 Gt.set_block(i, Gt_block);
771 }
772
773 return Gt;
774 }

797 template<bool isComplex>
798 dcomplex expectation_value(
799 triqs::operators::many_body_operator_real const &op,
800 triqs::atom_diag::atom_diag<isComplex> const &ad,
801 triqs::gfs::block_gf_view<triqs::mesh::dlr_imtime> G_ppsc) {
802
803 auto op_blocks = ad.get_op_mat(op);
804 auto beta = G_ppsc[0].mesh().beta();
805
806 dcomplex sum = 0;
807
808 for( auto bidx : range(op_blocks.block_mat.size()) ) {
809 assert( op_blocks.connection[bidx] == bidx );
810 if( op_blocks.block_mat[bidx].shape(0) == 0 ) continue; // skip empty blocks
811 auto g_dlr = make_gf_dlr(G_ppsc[bidx]);
812 auto U = ad.get_unitary_matrix(bidx);
813 auto op_mat_transf = U * op_blocks.block_mat[bidx] * nda::conj(nda::transpose(U));
814 sum += -trace(matmul(op_mat_transf, g_dlr(beta)));
815 }
816
817 return sum;
818 }
