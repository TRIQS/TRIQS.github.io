409 template <typename V>
410 requires(is_k_expr<V> or std::ranges::contiguous_range<V> or nda::ArrayOfRank<V, 1>)
411 [[nodiscard]] index_t closest_index(V const &k) const {
412 if constexpr (is_k_expr<V>) {
413 return closest_index(k.value());
414 } else {
415 // calculate k in the brzone basis
416 auto ks = nda::stack_vector<double, 3>{k[0], k[1], k[2]};
417 auto k_units = nda::transpose(units_inv_) * ks;
418 auto n = nda::stack_vector<long, 3>(nda::floor(k_units));
419
420 // calculate position relative to neighbors in mesh
421 auto w = k_units - n;
422
423 // prepare result container and distance measure
424 auto dst = std::numeric_limits<double>::infinity();
425
426 // check flatness along mesh dimensions
427 long r1 = std::min(dims_[0], 2l);
428 long r2 = std::min(dims_[1], 2l);
429 long r3 = std::min(dims_[2], 2l);
430
431 // find nearest neighbor by comparing distances
432 nda::stack_vector<long, 3> res;
433 for (auto const &[i1, i2, i3] : itertools::product_range(r1, r2, r3)) {
434 auto iv = nda::stack_vector<long, 3>{i1, i2, i3};
435 auto dstp = nda::linalg::norm(nda::transpose(units_) * (w - iv));
436
437 // update result when distance is smaller than current
438 if (dstp < dst) {
439 dst = dstp;
440 res = n + iv;
441 }
442 }
443
444 // fold back to brzone mesh (nearest neighbor could be out of bounds)
445 return index_modulo({res[0], res[1], res[2]});
446 }
447 }

534 std::cout << "WARNING: Reading old MeshBrillouinZone without BrillouinZone\n";
535 }

537 m = brzone(bz, dims);

560 // Helper function to do linear interpolation between k-points.
561 auto evaluate(detail::brzone1d const &m, auto const &f, double vi) {
562 long i = static_cast<long>(std::floor(vi));
563 double w = vi - double(i);
564 return (1 - w) * f(positive_modulo(i, m.dim)) + w * f(m.dim == 1 ? 0 : positive_modulo(i + 1, m.dim));
565 }
566
580 auto evaluate(brzone const &m, auto const &f, brzone::index_t const &n_tilde) { return f(m.index_modulo(n_tilde)); }

597 template <typename V>
598 requires(std::ranges::contiguous_range<V> or nda::ArrayOfRank<V, 1> or is_k_expr<V>)
599 auto evaluate(brzone const &m, auto const &f, V const &k) {
600 if constexpr (is_k_expr<V>) {
601 return evaluate(m, f, k.value());
602 } else {
603 auto v_index = nda::make_regular(nda::transpose(m.units_inv()) * nda::basic_array_view{k});
604 auto g = [&f](long x, long y, long z) { return f(typename brzone::index_t{x, y, z}); };
605 auto [d0, d1, d2] = m.dims();
606 return evaluate(std::tuple{detail::brzone1d{d0}, detail::brzone1d{d1}, detail::brzone1d{d2}}, g, v_index[0], v_index[1], v_index[2]);
607 }
608 }
