390 h5::assert_hdf5_format(gr, m, true); // NOLINT (downcasting to base class)
391
392 std::array<long, 3> dims{};
393 if (gr.has_key("dims")) {
394 h5::read(gr, "dims", dims);
395 } else {
396 // for backward compatibility
397 auto M = h5::read<nda::matrix<long>>(gr, "periodization_matrix");
398 dims = {M(0, 0), M(1, 1), M(2, 2)};
399 }
400
401 bravais_lattice bl{};
402 if (gr.has_key("bravais_lattice")) {
403 h5::read(gr, "bravais_lattice", bl);
404 } else {
405 // for backward compatibility
406 h5::read(gr, "bl", bl);
407 }
408
409 m = cyclat(bl, dims);
410 }
